<script>

/* =========================
   SURPRISE BUTTON
========================= */

const button =
document.getElementById("surpriseBtn");

const surprise =
document.getElementById("surprise");


button.addEventListener(
"click",
function() {

    surprise.classList.toggle("open");

    if (
        surprise.classList.contains("open")
    ) {

        button.innerHTML =
        "My Message For You ❤️";

        heartBurst();

    } else {

        button.innerHTML =
        "Open Your Surprise 💌";

    }

});


/* =========================
   FLOATING HEARTS
========================= */

function createHeart() {

    const heart =
    document.createElement("div");

    heart.className = "heart";

    const emojis = [
        "♥",
        "♡",
        "💗",
        "💕",
        "✨",
        "🌸"
    ];

    heart.innerHTML =
    emojis[
        Math.floor(
            Math.random() *
            emojis.length
        )
    ];

    heart.style.left =
    Math.random() * 100 + "vw";

    heart.style.fontSize =
    (12 + Math.random() * 20)
    + "px";

    heart.style.animationDuration =
    (5 + Math.random() * 5)
    + "s";

    document.body.appendChild(
        heart
    );

    setTimeout(
        () => heart.remove(),
        11000
    );
}


/* Create hearts continuously */

setInterval(
    createHeart,
    800
);


/* =========================
   HEART BURST
========================= */

function heartBurst() {

    for (
        let i = 0;
        i < 30;
        i++
    ) {

        setTimeout(
            () => {

                const heart =
                document.createElement(
                    "div"
                );

                heart.className =
                "heart";

                heart.innerHTML =
                ["💖","💗","💕","✨"]
                [
                    Math.floor(
                        Math.random() * 4
                    )
                ];

                heart.style.left =
                (35 +
                Math.random() * 30)
                + "vw";

                heart.style.bottom =
                (20 +
                Math.random() * 20)
                + "vh";

                heart.style.animationDuration =
                (2 +
                Math.random() * 2)
                + "s";

                document.body.appendChild(
                    heart
                );

                setTimeout(
                    () => heart.remove(),
                    5000
                );

            },
            i * 50
        );

    }

}

    ")";

    document.body.appendChild(
        sparkle
    );
}
